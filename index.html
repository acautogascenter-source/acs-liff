async function initLine() { try { await liff.init({ liffId: LIFF_ID }); if (!liff.isLoggedIn()) { const note = $('lineStatus'); if (note) { note.textContent = 'กรุณาเปิดผ่าน LINE และเข้าสู่ระบบเพื่อรับข้อความยืนยันการจอง'; } // ถ้าอยู่ใน LIFF ให้เข้าสู่ระบบ LINE if (liff.isInClient()) { liff.login({ redirectUri: location.href }); } return; } const profile = await liff.getProfile(); lineUserId = profile.userId || ''; if (!lineUserId) { throw new Error('ไม่สามารถอ่าน LINE User ID ได้'); } const saved = await api('saveLineUser', { userId: lineUserId, displayName: profile.displayName || '', source: 'acs-liff' }); if (!saved.success) { throw new Error(saved.message || 'บันทึก LINE User ID ไม่สำเร็จ'); } console.log('LINE CONNECTED:', lineUserId); } catch (err) { console.error('LIFF ERROR:', err); lineUserId = ''; } }
